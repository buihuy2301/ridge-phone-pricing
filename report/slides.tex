\section{Đặt vấn đề}

\begin{frame}{Bài toán và dữ liệu}
  \begin{itemize}
    \item Dữ liệu định giá điện thoại đã qua sử dụng (Kaggle):
          \num{1000000} bản ghi, 28 cột, không có giá trị thiếu.
    \item 21 cột định lượng, 6 cột định tính (\texttt{brand} 7 mức,
          \texttt{model} 23 mức, \texttt{condition} 4 mức, \dots).
    \item Thí nghiệm chạy trên mẫu \num{200000} bản ghi: $n = \num{160000}$
          điểm huấn luyện, $d = \num{280}$ thuộc tính sau mã hóa.
    \item Biến mục tiêu lấy logarit vì giá bán lại lệch phải rõ rệt.
    \item Mục tiêu của bài tập là phần tối ưu hóa, không phải phần mô hình hóa.
  \end{itemize}
  \note{Nhắc lại rằng hàm mục tiêu được cố định trước, mọi so sánh về sau đều
  trên cùng một hàm. Lấy mẫu là quyết định về chi phí tính toán: với một triệu
  dòng, một lần tính gradient mất 271 ms thay vì 39 ms.}
\end{frame}

\begin{frame}{Bài toán tối ưu hóa}
  \[
    \min_{w \in \R^{d}} \;
    f(w) = \frac{1}{2n}\norm{Xw-y}_{2}^{2} + \frac{\lambda}{2}\norm{w}_{2}^{2}
  \]
  \[
    \nabla f(w) = \frac{1}{n}X\trans(Xw-y) + \lambda w,
    \qquad
    \nabla^{2} f(w) = \frac{1}{n}X\trans X + \lambda I
  \]
  \[
    \wstar = \Bigl(\tfrac{1}{n}X\trans X + \lambda I\Bigr)^{-1}
             \Bigl(\tfrac{1}{n}X\trans y\Bigr)
  \]
  Có nghiệm đóng, nên $\fstar$ tính được chính xác và mọi biểu đồ vẽ
  $f(w_k) - \fstar$ trên thang logarit.
\end{frame}

\section{Đặc trưng của bài toán}

\begin{frame}{Các hằng số quyết định tốc độ hội tụ}
  \[
    L = \lambda_{\max}\Bigl(\tfrac{1}{n}X\trans X\Bigr) + \lambda,
    \quad
    \mu = \lambda_{\min}\Bigl(\tfrac{1}{n}X\trans X\Bigr) + \lambda,
    \quad
    \kappa = L/\mu
  \]
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \begin{tabular}{lr}
        \toprule
        $L$        & \num{2.683} \\
        $\mu$      & \num{0.01} \\
        $\kappa$   & \num{268.3} \\
        $\lambda$  & \num{0.01} \\
        $\fstar$   & \num{0.0237430} \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.54\textwidth}
      Ma trận Gram có hạng 273 trên 280, trị riêng kế tiếp cỡ $2 \cdot 10^{-8}$.
      \medskip

      Do đó $\mu = \lambda$ và $\kappa = L / \lambda$: hệ số hiệu chỉnh là
      đại lượng duy nhất quyết định số điều kiện.
    \end{column}
  \end{columns}
  \note{Đây là điểm cần nắm chắc: mọi câu hỏi về vì sao kappa lại như vậy đều
  quay về chỗ Gram suy biến, không phải do dữ liệu xấu mà do các cột phụ thuộc
  tuyến tính.}
\end{frame}

\begin{frame}{Chọn $\lambda$: cross-validation không phân biệt được}
  \begin{itemize}
    \item Đường cong CV 5 fold phẳng từ $\lambda = 10^{-6}$ đến $10^{-3}$,
          sai số giống nhau tới bốn chữ số thập phân.
    \item Cùng khoảng đó, $\kappa = L/\lambda$ chênh nhau 1000 lần.
    \item Lấy điểm cực tiểu là lựa chọn tùy tiện. Dùng quy tắc một sai số
          chuẩn: chọn $\lambda$ lớn nhất còn nằm trong một sai số chuẩn của
          giá trị tốt nhất.
    \item Kết quả $\lambda = \num{0.01}$, $\kappa = \num{268.3}$.
          Giá phải trả: RMSE test \num{0.20601} so với \num{0.20579}, kém 0{,}1\%.
  \end{itemize}
\end{frame}

\begin{frame}{Kết luận về chọn độ dài bước cho GD}
  \begin{itemize}
    \item Trong vùng ổn định, số vòng lặp tỉ lệ nghịch với độ dài bước:
          20, 40, 80 vòng ứng với $t = 1.9/L$, $1/L$, $0.5/L$.
    \item Nên chọn $t$ càng gần $2/L$ càng tốt. $t = 2/(L+\mu)$ tốt nhất trong
          lưới nhưng gần như trùng $2/L$ vì $\mu \ll L$.
    \item Backtracking chậm hơn bước cố định tốt nhất 10\%
          (\SI{0.852}{\second} so với \SI{0.773}{\second}), đổi lại không cần

  \end{itemize}
\end{frame}

\begin{frame}{SGD: độ dài bước phải theo kích thước lô}
  \[
    L = \num{2.683}, \qquad L_{\max} = \max_i \norm{x_i}^{2} + \lambda = \num{3109.7}
  \]
  Chênh nhau 1159 lần. Ngưỡng ổn định của một bước ngẫu nhiên do độ cong của các
  mẫu trong lô quyết định, không do $L$.
  \[
    L_B = L + \frac{L_{\max} - L}{B}, \qquad \eta_0 = 1/L_B
  \]
\end{frame}

\begin{frame}{AGD: backtracking cần điều kiện chặt hơn}
  Line search chạy tại điểm ngoại suy $y_k$, nên chỉ bảo đảm
  $f(w_{k+1}) < f(y_k)$, không bảo đảm đơn điệu theo $w$. Armijo có thể chấp
  nhận $t$ tới $1/\mu = 100$ trong khi ngưỡng ổn định là $2/L = \num{0.745}$.
  \begin{itemize}
    \item $\alpha = 0.3$, $t_0 = 1$: không hội tụ, dừng ở $\num{5.5e-5}$.
    \item $\alpha = 0.5$: điều kiện Armijo trùng bổ đề giảm
          $f(y - t\nabla f(y)) \le f(y) - \tfrac{t}{2}\norm{\nabla f(y)}^{2}$. Hội tụ.
    \item $t_0 = 1/L$: line search chỉ thu nhỏ từ giá trị vốn an toàn. Hội tụ.
  \end{itemize}
  Momentum cũng phải theo bước thực tế:
  $\beta = \bigl(1 - \sqrt{t\mu}\bigr) / \bigl(1 + \sqrt{t\mu}\bigr)$.
  \note{Công thức quen thuộc theo căn kappa chỉ là trường hợp riêng t = 1/L.
  Ghép sai làm hàm mục tiêu bùng lên 1.47e4.}
\end{frame}

\begin{frame}{Phương pháp Newton}
  \[
    w_1 = w_0 - \bigl(\nabla^{2} f\bigr)^{-1}\nabla f(w_0) = \wstar
  \]
  Với hàm bậc hai, một vòng lặp là đủ và bước Newton chính là nghiệm đóng.
  Chi phí một vòng lặp là $\bigO(nd^{2} + d^{3})$ so với $\bigO(nd)$ của
  gradient descent.
  \begin{center}\resultgraphic[0.5]{newton_time}\end{center}
\end{frame}

\section{So sánh với thư viện}

\begin{frame}{Quy đổi hàm mục tiêu}
  \begin{itemize}
    \item \texttt{Ridge} cực tiểu hóa $\norm{Xw-y}_{2}^{2} + \alpha\norm{w}_{2}^{2}$,
          nên $\alpha = \lambda n$.
    \item \texttt{SGDRegressor} cực tiểu hóa đúng dạng \eqref{eq:objective-slide},
          nên $\alpha = \lambda$.
  \end{itemize}
  \begin{equation}\label{eq:objective-slide}
    f(w) = \frac{1}{2n}\norm{Xw-y}_{2}^{2} + \frac{\lambda}{2}\norm{w}_{2}^{2}
  \end{equation}
  Hai quy đổi được kiểm chứng bằng số, không suy luận trên giấy.
\end{frame}

\section{Kết luận}

\begin{frame}{Ảnh hưởng của $\lambda$}

    \item Nhưng $\kappa$ giảm 96 lần chỉ đổi được số vòng lặp từ 40 xuống 30,

          chiều.
  \end{itemize}

\begin{frame}{Những điều rút ra}
  \begin{itemize}
    \item \textbf{Độ dài bước:} $2/L$ là ranh giới thật, chênh 10\% là khác biệt
          giữa hội tụ và phân kỳ. Trong vùng ổn định, chọn càng gần $2/L$ càng tốt.
    \item \textbf{Backtracking:} $\beta$ quyết định chi phí, $\alpha$ gần như
          không. Với phương pháp tăng tốc thì cần $\alpha = 1/2$ hoặc
          $t_0 = 1/L$, và momentum phải tính theo bước thực tế.
    \item \textbf{SGD:} độ dài bước phải theo kích thước lô;
          $L_{\max}/L = 1159$ trên bộ dữ liệu này.
    \item \textbf{Bậc hai:} với hàm bậc hai, Newton là nghiệm đóng nên vượt trội
          hoàn toàn. Kết luận này không tổng quát hóa sang bài toán phi tuyến.
    \item \textbf{Lý thuyết và thực nghiệm:} cận chặt ở phần đuôi, lỏng ở giai
          đoạn đầu, vì phổ trị riêng của bài toán tập trung.
  \end{itemize}
\end{frame}
