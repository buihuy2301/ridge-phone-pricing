\begin{frame}{Hàm mục tiêu và các hằng số quyết định tốc độ hội tụ}
  \[
    \min_{w \in \R^{d}} \;
    f(w) = \frac{1}{2n}\norm{Xw-y}_{2}^{2} + \frac{\lambda}{2}\norm{w}_{2}^{2},
    \quad
    L = \lambda_{\max}\Bigl(\tfrac{1}{n}X\trans X\Bigr) + \lambda,
    \;
    \mu = \lambda_{\min}\Bigl(\tfrac{1}{n}X\trans X\Bigr) + \lambda,
    \;
    \kappa = L/\mu
  \]
  \begin{columns}[T]
    \begin{column}{0.38\textwidth}
      \begin{tabular}{lr}
        \toprule
        $L$        & \num{2.683} \\
        $\mu$      & \num{0.01} \\
        $\kappa$   & \num{268.3} \\
        $\lambda$  & \num{0.01} \\
        $\fstar$   & \num{0.0237430} \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.58\textwidth}
      Hessian $\tfrac{1}{n}X\trans X + \lambda I$ không phụ thuộc $w$, nên bài
      toán lồi mạnh và có nghiệm đóng $\wstar$. Nhờ vậy $\fstar$ tính được trước
      khi chạy thuật toán nào, và mọi biểu đồ vẽ được sai số tuyệt đối
      $f(w_k) - \fstar$ trên thang logarit.
      \medskip

      Ma trận Gram có hạng 273 trên 280 và trị riêng kế tiếp chỉ cỡ
      $2 \cdot 10^{-8}$, tức phần đóng góp của dữ liệu vào $\mu$ coi như bằng
      không, nên $\mu = \lambda$ và $\kappa = L/\lambda$: hệ số hiệu chỉnh là đại
      lượng duy nhất quyết định số điều kiện.
    \end{column}
  \end{columns}
  \note{\textbf{2:45--4:30.} Vế đầu của $f$ là trung bình bình phương sai số, vế
  sau phạt trọng số lớn. Chia cho $n$ để $L$ không phụ thuộc số mẫu, nhờ vậy đặt
  cạnh SGD mới công bằng, còn chia thêm cho 2 là để số 2 triệt tiêu khi lấy đạo
  hàm.

  Hessian không phụ thuộc $w$, tức chỗ nào cũng cong như nhau, nên mặt hàm mục
  tiêu là một cái bát parabol và bài toán có nghiệm đóng. Nhờ đó nhóm em biết
  trước $\fstar = \num{0.0237430}$, và mọi biểu đồ vẽ được sai số tuyệt đối
  $f(w_k) - \fstar$ trên thang logarit. Không có $\fstar$ thì chỉ so được với
  giá trị tốt nhất từng thấy, và đường nào cũng cắm xuống ở cuối biểu đồ, che
  mất hành vi thật.

  Ba con số trong bảng: $L$ là độ cong lớn nhất, $\mu$ là độ cong nhỏ nhất,
  $\kappa$ là độ méo của cái bát. Bát méo 268 lần nghĩa là nó bị bóp thành một
  cái máng dài, hòn bi lăn ngang thì nhanh mà bò dọc theo máng thì chậm hơn
  nhiều bậc, và toàn bộ khó khăn của nhóm bậc một nằm ở đó.

  Chỗ phải nói được là $\mu = \lambda$. Ma trận Gram có hạng 273 trên 280, trị
  riêng kế tiếp chỉ cỡ $2 \cdot 10^{-8}$, nhỏ hơn $L$ tám bậc, nên phần đóng góp
  của dữ liệu vào $\mu$ coi như bằng không. Vậy $\kappa = L/\lambda$: hệ số hiệu
  chỉnh quyết định số điều kiện, dữ liệu không có tiếng nói.

  \textbf{Nếu bị hỏi khi nào kết luận đó đảo chiều:} khi mọi cột độc lập tuyến
  tính và trị riêng nhỏ nhất cùng bậc với $\lambda$, khi ấy $\mu$ quay về do dữ
  liệu quyết định.}
\end{frame}

\section{Khảo sát tham số}

\begin{frame}{Gradient descent: độ dài bước cố định}
  \begin{center}\resultgraphic[0.62]{gd_fixed_iter}\end{center}
  Chênh 10\% quanh ngưỡng $2/L = \num{0.745}$ đã đổi hẳn kết cục: $t = 2.1/L$
  đẩy hàm mục tiêu lên $\num{5.4e11}$, còn $t = 1.9/L$ hội tụ sau 20 vòng lặp.
  \note{\textbf{4:30--6:00.} Gradient descent chỉ có một tham số là độ dài bước
  $t$. Bước ngắn quá thì lâu mới tới đáy, bước dài quá thì vọt qua đáy sang sườn
  bên kia, có khi lên cao hơn cả chỗ vừa đứng.

  Mời thầy cô nhìn đường đi lên ở phía trên biểu đồ. Đó là $t = 2.1/L$, nhóm em
  cố ý cho chạy vượt ngưỡng, và nó đẩy hàm mục tiêu lên $\num{5.4e11}$. Ngay
  cạnh đó, $t = 1.9/L$ hội tụ sau 20 vòng lặp. Hai cấu hình chênh nhau 10\%
  quanh $2/L = \num{0.745}$ mà kết cục lật hẳn.

  Cơ chế nằm ở từng hướng riêng. Sau mỗi vòng, sai số theo hướng ứng với trị
  riêng $\lambda_i$ nhân với $\abs{1 - t\lambda_i}$, nên muốn tắt dần thì hệ số
  ấy phải nhỏ hơn 1. Hướng dốc nhất có $\lambda_i = L$, do đó khi $t$ vượt $2/L$
  thì $\abs{1 - tL} > 1$ và thành phần đó nhân lên mỗi vòng thay vì tắt đi. Đúng
  tại ngưỡng, hệ số bằng 1, tức thành phần đó đứng im mãi, và vì vậy $t = 2/L$
  kẹt ở $\num{4.9e-8}$ chứ không đi tiếp được.

  Trong vùng an toàn thì chia đôi $t$ làm số vòng lặp nhân đôi, thấy rõ ở dãy
  20, 40, 80 ứng với $1.9/L$, $1/L$, $0.5/L$. Kết luận thực dụng: chọn $t$ càng
  sát $2/L$ càng tốt.

  \textbf{Nếu bị hỏi về $t = 2/(L+\mu)$:} đó là giá trị tối ưu về lý thuyết,
  nhưng ở đây nó chỉ hơn $1.9/L$ có 0,07 giây, vì $\mu \ll L$ nên
  $2/(L+\mu) \approx 2/L$.}
\end{frame}

\begin{frame}{Gradient descent: backtracking và cách chọn độ dài bước}
  \begin{center}
    \small
    \begin{tabular}{lrrr}
      \toprule
      $\alpha = 0.5$ & Đánh giá hàm / vòng lặp & Vòng lặp đạt $10^{-6}$ & Thời gian (s) \\
      \midrule
      $\beta = 0.5$ & \num{1.39} & 15 & \num{0.852} \\
      $\beta = 0.8$ & \num{2.37} & 20 & \num{1.617} \\
      $\beta = 0.9$ & \num{3.96} & 20 & \num{1.799} \\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{itemize}
    \item $\beta$ quyết định chi phí, còn $\alpha$ gần như không ảnh hưởng, vì
          ba giá trị $\alpha \in \{0.1, 0.3, 0.5\}$ chỉ chênh nhau 4{,}8\% về số
          lần đánh giá hàm.
    \item Trong vùng ổn định, số vòng lặp tỉ lệ nghịch với độ dài bước, 20, 40,
          80 vòng ứng với $t = 1.9/L$, $1/L$, $0.5/L$, nên chọn $t$ càng gần
          $2/L$ càng tốt.
    \item Backtracking chậm hơn bước cố định tốt nhất 10\%
          (\SI{0.852}{\second} so với \SI{0.773}{\second}), và khoản chênh đó mua
          lấy chỗ không phải biết $L$.
  \end{itemize}
  \note{\textbf{6:00--7:45.} Bước cố định đòi biết $L$ trước. Không tính được
  $L$ thì thử: bắt đầu bằng một bước dài, thấy chưa giảm đủ thì nhân với
  $\beta < 1$, lặp tới khi điều kiện Armijo chấp nhận. Nói nôm na, điều kiện ấy
  bắt bước hiện tại phải giảm được ít nhất $\alpha$ phần của mức giảm mà độ dốc
  hứa hẹn.

  Cột giữa của bảng là số lần đánh giá hàm mục tiêu trên mỗi vòng lặp, đo trong
  100 vòng đầu. Đọc dọc thì $\beta$ quyết định chi phí: $\beta = 0.5$ tốn
  \num{1.39} lần, $\beta = 0.9$ tốn \num{3.96} lần, vì $\beta$ càng gần 1 thì
  mỗi lần rút ngắn càng ít nên phải thử nhiều lần hơn. Đọc ngang thì đổi
  $\alpha$ gần như không đổi gì, chênh nhiều nhất 4,8\%.

  Cặp $\beta$ này còn là ví dụ sạch nhất trong cả bài cho việc vì sao đề bài bắt
  vẽ hai biểu đồ.
  $\beta = 0.8$ và $\beta = 0.9$ gần trùng nhau trên trục vòng lặp nhưng tách
  hẳn trên trục thời gian, vì chênh lệch nằm ở số lần đánh giá hàm bên trong mỗi
  vòng, mà trục vòng lặp không nhìn thấy chi phí đó.

  Chốt lại, backtracking chậm hơn bước cố định tốt nhất 10\%, và khoản chênh đó
  mua lấy chỗ không phải biết $L$. Bài này $L$ tính dễ nên bước cố định hợp lý
  hơn, nhưng nhận xét ấy đảo chiều ngay khi $d$ lớn tới mức không tính nổi trị
  riêng lớn nhất của ma trận Gram.

  \textbf{Nếu bị hỏi vì sao chỉ lấy 100 vòng đầu:} phép đo phải nằm trong giai
  đoạn thuật toán còn tiến triển. Khi $f$ đã chạm giới hạn số học, Armijo không
  thỏa mãn được với bất kỳ bước nào và con số bị thổi lên gấp năm lần.}
\end{frame}

\begin{frame}{SGD: độ dài bước theo kích thước lô}
  \[
    L = \num{2.683}, \quad
    L_{\max} = \max_i \norm{x_i}^{2} + \lambda = \num{3109.7}
    \qquad\Longrightarrow\qquad
    L_B = L + \frac{L_{\max} - L}{B}, \quad \eta_0 = 1/L_B
  \]
  \begin{center}\resultgraphic[0.44]{sgd_common_step_epoch}\end{center}
  \begin{minipage}{0.9\textwidth}
    Dùng chung $\eta = 0.1/L$ thì lô kích thước 1 phân kỳ thành \texttt{NaN}
    ngay trong lượt duyệt (epoch) đầu tiên, còn lô 1024 đạt sai số
    $\num{1.1e-4}$. Ngưỡng ổn định của một bước ngẫu nhiên phụ thuộc độ cong của
    các mẫu trong lô chứ không phụ thuộc $L$, mà hai hằng số này chênh nhau 1159
    lần, nên chỉ lô lớn mới nằm trong vùng ổn định của một bước tính theo $L$.
  \end{minipage}
  \note{\textbf{7:45--9:15.} Mỗi vòng gradient descent phải quét cả \num{160000}
  dòng để tính một gradient. SGD chỉ lấy ngẫu nhiên $B$ dòng rồi bước luôn, nên
  hướng đi lệch lạc mà số bước đi được trong cùng thời gian nhiều hơn hẳn.

  Cái bẫy nằm ở hai hằng số trên đầu slide. $L$ của cả hàm là \num{2.683}, còn
  $L_{\max}$, tức $L$ của mẫu đơn lẻ dốc nhất, đo được \num{3109.7}, lớn hơn
  1159 lần. Nghĩa là có những dòng dữ liệu mà riêng nó tạo ra một cái bát dốc
  đứng gấp nghìn lần cái bát trung bình.

  Nhóm em làm thí nghiệm đối chứng để chỉ ra chỗ đó: dùng chung
  $\eta = 0.1/L$ cho mọi kích thước lô thì $B = 1$ phân kỳ thành \texttt{NaN}
  ngay trong lượt duyệt đầu tiên, còn $B = 1024$ vẫn về được $\num{1.1e-4}$.
  Ngưỡng ổn định của một bước ngẫu nhiên phụ thuộc độ cong của các mẫu trong lô
  chứ không phụ thuộc $L$, nên chỉ lô lớn mới nằm trong vùng an toàn của một
  bước tính theo $L$.

  Cách xử lý là cho mỗi kích thước lô một bước riêng, $\eta_0 = 1/L_B$. Công
  thức $L_B$ nội suy giữa hai đầu: $B = 1$ thì $L_B \approx L_{\max}$, còn $B$
  lớn thì $L_B \to L$.

  \textbf{Nếu bị hỏi vì sao SGD không có biến thể backtracking:} hai lý do. Một,
  quá đắt, vì Armijo cần đánh giá $f$ trên toàn bộ dữ liệu ở mỗi lần thử bước,
  mà với $B = 64$ thì một lượt duyệt có 2500 lần cập nhật, ước lượng 98 giây cho
  một lượt duyệt so với 0,082 giây đo được. Hai, không bảo đảm gì, vì Armijo áp
  lên gradient của lô chỉ bảo đảm giảm cho hàm mất mát của lô đó.}
\end{frame}

\begin{frame}{Accelerated gradient descent}
  \begin{center}\resultgraphic[0.42]{agd_iter}\end{center}
  \begin{minipage}{0.9\textwidth}
    Khởi động lại không đổi số vòng lặp đạt $10^{-6}$, nhưng giảm tổng thời gian
    từ \SI{10.5}{\second} xuống \SI{4.86}{\second} vì nó khử dao động ở phần
    đuôi.
    \begin{itemize}
      \item Với $\alpha = 0.3$ và $t_0 = 1$, phương pháp không hội tụ mà dừng ở
            $\num{5.5e-5}$, vì line search chạy tại điểm ngoại suy $y_k$ nên chỉ
            bảo đảm $f(w_{k+1}) < f(y_k)$ và Armijo chấp nhận $t$ tới
            $1/\mu = 100$ trong khi ngưỡng ổn định là $2/L = \num{0.745}$.
      \item Hai cách xử lý đều đưa phương pháp về hội tụ, lấy $\alpha = 0.5$ để
            điều kiện Armijo trùng bổ đề giảm, hoặc khởi tạo $t_0 = 1/L$ để line
            search chỉ thu nhỏ từ một giá trị vốn an toàn. Momentum phải tính
            theo bước thực tế,
            $\beta = \bigl(1 - \sqrt{t\mu}\bigr)/\bigl(1 + \sqrt{t\mu}\bigr)$.
    \end{itemize}
  \end{minipage}
  \note{\textbf{10:45--12:15.} Gradient descent như hòn bi lăn trong máng đầy
  mật, mỗi bước chỉ nghe theo độ dốc tại chỗ. AGD cho hòn bi có quán tính: nó
  nhớ hướng vừa đi và lao thêm theo hướng đó một chút. Nesterov khác momentum
  thường ở chỗ trượt tới điểm ngoại suy $y_k$ trước rồi mới lấy gradient ở đó.
  Về lý thuyết, hệ số co rút cải thiện từ $1 - \bigO(1/\kappa)$ lên
  $1 - \bigO(1/\sqrt{\kappa})$, với $\kappa = 268$ thì nhanh hơn cỡ 16 lần ở
  phần đuôi.

  Một kết quả ngược lý thuyết đáng nói: công thức
  $\beta = (\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$ vốn tối ưu cho hàm lồi mạnh lại
  cần 35 vòng, còn công thức tăng dần $\beta_k = (k-1)/(k+2)$ chỉ cần 20. Vì
  công thức hằng nhắm vào trường hợp xấu nhất trong lớp hàm có $\kappa$ cho
  trước, tức phổ trị riêng trải đều trên $[\mu, L]$, mà bài toán này có phổ dồn
  cụm nên bộ tham số phòng thủ ấy mất lợi thế.

  Momentum tăng dần gây dao động tuần hoàn, nhìn thấy như gợn sóng ở đuôi biểu
  đồ. Khởi động lại là khi phát hiện $\nabla f(y_k)\trans(w_{k+1}-w_k) > 0$, tức
  đang đi ngược hướng dốc, thì xả hết quán tính và tích lại từ đầu. Nó không đổi
  số vòng lặp đạt $10^{-6}$ nhưng giảm tổng thời gian từ \SI{10.5}{\second}
  xuống \SI{4.86}{\second}.

  \textbf{Nên kể chỗ này:} lần cài đặt đầu, nhóm em để line search chọn $t$
  nhưng vẫn tính momentum theo $1/L$, và hàm mục tiêu bùng lên $\num{1.47e4}$.
  Chương trình chạy trót lọt, không báo lỗi, chỉ trả về một con số trông hợp lý,
  nên chỉ có đối chiếu với mốc giải tích mới bắt được loại lỗi này.}
\end{frame}

\begin{frame}{Phương pháp Newton}
  \[
    w_1 = w_0 - \bigl(\nabla^{2} f\bigr)^{-1}\nabla f(w_0) = \wstar
  \]
  Với hàm bậc hai, một vòng lặp là đủ và bước Newton chính là nghiệm đóng, nên
  toàn bộ lời giải mất \SI{0.076}{\second}. Đại lượng đáng so ở đây vì thế là
  chi phí một vòng lặp, $\bigO(nd^{2} + d^{3})$ so với $\bigO(nd)$ của gradient
  descent, chứ không phải số vòng lặp.
  \begin{center}\resultgraphic[0.5]{newton_time}\end{center}
  \note{\textbf{12:15--13:20.} Gradient descent chỉ biết hướng dốc, Newton biết
  cả độ cong nên không mò từng bước mà nhảy thẳng tới đáy của xấp xỉ bậc hai tại
  chỗ đang đứng. Ở đây $f$ vốn là hàm bậc hai, xấp xỉ bậc hai của nó chính là
  nó, nên xuất phát từ $w_0 = 0$ thì một vòng lặp ra đúng $\wstar$. Nói cách
  khác, bước Newton chính là nghiệm đóng.

  Vì vậy đại lượng đáng so ở đây là chi phí một vòng lặp chứ không phải số vòng
  lặp: $\bigO(nd^2 + d^3)$ so với $\bigO(nd)$ của gradient descent. Toàn bộ lời
  giải mất \SI{0.076}{\second}.

  Ba biến thể Newton trên biểu đồ gần như trùng nhau, và cần giải thích đúng lý
  do: backtracking chấp nhận ngay $t = 1$ nên chỉ tốn một lần đánh giá hàm, còn
  phân rã lại hay dùng lại thì vô nghĩa khi chỉ có một vòng lặp. Hai cặp này chỉ
  tách nhau trên bài toán phi tuyến.

  \textbf{Nếu bị hỏi vì sao dùng Cholesky chứ không phải \texttt{np.linalg.inv}:}
  nghịch đảo tường minh kém ổn định về số học và tốn nhiều phép tính hơn phân rã
  rồi giải thế.

  \textbf{Nếu bị hỏi vì sao Newton-CG chậm hơn Newton 25 lần:} vì $n \gg d$ ở
  bài này. Một lần Cholesky tốn $\bigO(d^3) \approx 2 \cdot 10^{7}$, còn mỗi
  tích Hessian nhân vector tốn $\bigO(nd) \approx 4.5 \cdot 10^{7}$, mà CG cần
  nhiều tích như vậy. Quan hệ chỉ đảo khi $d$ lớn tới mức không lập nổi ma trận
  $d \times d$.}
\end{frame}

  $\num{1.06e-14}$, và chênh lệch hàm mục tiêu bằng đúng 0.

  Hai dòng đáng nói trong bảng. Solver trực tiếp đạt $\fstar$ với sai số cỡ sai
  số máy trong \SI{0.112}{\second}, vì nó giải thẳng chính cái hệ mà một bước
  Newton giải. Còn \texttt{SGDRegressor} ở tham số mặc định dừng cách $\fstar$
  tận $\num{1.16e-3}$, gấp 200 lần cấu hình SGD nhóm em tự cài, vì quy tắc chọn
  bước mặc định của nó không tính theo $L_B$ nên rơi đúng vào cái lân cận đã nói
  ở phần SGD. Cần nói rõ giới hạn: nhận xét này áp cho bộ ước lượng đó ở tham số
  mặc định, không áp cho scikit-learn nói chung.

\begin{frame}{Ảnh hưởng của $\lambda$}
  \begin{center}
    \begin{tabular}{rrrrr}
      \toprule
      $\lambda$ & $\kappa$ & Vòng lặp đạt $10^{-6}$ & Thời gian (s) & RMSE test \\
      \midrule
      \num{0.001} & \num{2674}  & 40 & \num{1.756} & \num{0.20580} \\
      \num{0.01}  & \num{268.3} & 40 & \num{1.771} & \num{0.20601} \\
      \num{0.1}   & \num{27.73} & 30 & \num{1.233} & \num{0.21916} \\
      \bottomrule
    \end{tabular}
  \end{center}
  \begin{itemize}
    \item Vì $\mu = \lambda$, tăng $\lambda$ mười lần thì giảm $\kappa$ đúng
          mười lần.
    \item Nhưng $\kappa$ giảm 96 lần chỉ đổi được số vòng lặp từ 40 xuống 30,
          trong khi RMSE xấu đi 6{,}5\%, vì giai đoạn đầu do các hướng có độ cong
          lớn chi phối chứ không do $\kappa$.
    \item Hiệu chỉnh vì thế có hai vai trò không cùng chiều, thống kê và tối ưu
          hóa, và đánh đổi giữa chúng chỉ có lợi khi cần độ chính xác cao.
  \end{itemize}
  \note{\textbf{Dùng khi bị hỏi $\lambda$ dính gì tới tốc độ hội tụ.} Slide này
  nối chủ đề Ridge với nội dung môn học, nên nếu còn thời gian ở phần chính thì
  cũng đáng chiếu. Vì $\mu = \lambda$ trên bài toán này, tăng $\lambda$ mười lần
  là giảm $\kappa$ đúng mười lần, tức hệ số hiệu chỉnh có luôn một vai trò tối
  ưu hóa bên cạnh vai trò thống kê quen thuộc.

  Nhưng con số trong bảng nói rằng hai vai trò ấy lệch nhau rất xa. Từ
  $\lambda = 0.001$ lên $0.1$, $\kappa$ giảm 96 lần mà số vòng lặp chỉ giảm từ
  40 xuống 30, trong khi RMSE xấu đi 6,5\%. Số vòng lặp ít nhạy cảm như vậy vì
  cùng lý do đã nói ở phần so sánh tổng hợp, rằng giai đoạn đầu do các hướng có
  độ cong lớn chi phối chứ không do $\kappa$.

  Chốt: tăng $\lambda$ để cải thiện điều kiện là đánh đổi không có lợi nếu chỉ
  cần độ chính xác vừa phải, và nó chỉ đáng giá khi cần độ chính xác cao, nơi
  phần đuôi mới tốn kém.}
\end{frame}

\begin{frame}{Ba thuật toán cài đặt thêm}
  \begin{itemize}
    \item \textbf{Heavy ball.} Cùng dạng momentum như Nesterov nhưng ngoại suy
          sau khi lấy gradient:
          $w_{k+1} = w_k - t\nabla f(w_k) + \beta(w_k - w_{k-1})$, với
          $t = 4/(\sqrt{L}+\sqrt{\mu})^{2}$ và
          $\beta = \bigl((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)\bigr)^{2}$.
    \item \textbf{Newton-CG.} Giải hệ Newton bằng gradient liên hợp. Mỗi bước CG
          chỉ cần một tích Hessian nhân vector,
          $\tfrac{1}{n}X\trans(Xv) + \lambda v$, chi phí $\bigO(nd)$, nên không
          phải lập ma trận $d \times d$. Ở đây chậm hơn Newton 25 lần vì
          $d = 280$ còn nhỏ.
    \item \textbf{L-BFGS.} Xấp xỉ nghịch đảo Hessian từ $m$ cặp $(s_k, y_k)$ gần
          nhất qua đệ quy hai vòng, chi phí $\bigO(md)$ mỗi vòng lặp. Nằm giữa
          nhóm bậc một và bậc hai về cả chi phí lẫn tốc độ, đạt $10^{-6}$ sau 8
          vòng lặp và \SI{0.448}{\second}.
  \end{itemize}
  Adam cũng được cài đặt nhưng không đưa vào so sánh chính, vì trên hàm bậc hai
  lồi mạnh có nghiệm đóng, cơ chế chia theo căn bậc hai của moment bậc hai không
  mang lại lợi thế nào.
  \note{\textbf{Dùng khi bị hỏi ngoài bốn thuật toán bắt buộc còn gì.} Ba thuật
  toán này nằm ngoài yêu cầu của đề, nhóm em cài thêm để lấp khoảng giữa nhóm
  bậc một và nhóm bậc hai.

  Nếu bị hỏi riêng về L-BFGS thì có hai quan sát đáng kể. Chọn $m$ không quan
  trọng ở giai đoạn đầu, cả bốn cấu hình đạt $10^{-6}$ sau đúng 8 vòng lặp, vì
  trong 8 vòng đó số cặp tích lũy được còn ít hơn cả $m = 3$. Tới đuôi thì $m$
  mới có tác dụng, từ 54 vòng với $m = 3$ xuống 37 vòng với $m = 20$.

  Nhưng tổng thời gian không giảm đều theo $m$: $m = 10$ tốn \SI{3.95}{\second}
  trong khi $m = 3$ chỉ tốn \SI{3.18}{\second} dù cần nhiều hơn 14 vòng lặp.
  Chênh lệch đến từ line search, vốn tiêu 227 lần đánh giá hàm ở $m = 10$ so với
  105 lần ở $m = 5$. Với $d = 280$ thì chi phí $\bigO(md)$ nhỏ hơn hẳn một lần
  đánh giá $f$, nên line search mới là phần chi phối chứ không phải $m$.}
\end{frame}

\begin{frame}{Kiểm thử tính đúng đắn của cài đặt}
  46 phép thử tự động chạy trước khi khảo sát tham số. Bốn phép thử đối chiếu
  trực tiếp với công thức giải tích:
  \begin{itemize}
    \item gradient giải tích so với sai phân trung tâm tại
          $\varepsilon = 10^{-6}$, sai số tương đối dưới $10^{-6}$;
    \item Hessian so với sai phân của gradient, cùng ngưỡng;
    \item nghiệm đóng là điểm dừng, $\norm{\nabla f(\wstar)} < 10^{-12}$;
    \item mười cấu hình thuật toán cùng hội tụ về một nghiệm trên bài toán tổng
          hợp $n = 100$, $d = 5$.
  \end{itemize}
  Ba phép thử về mặt phương pháp luận: gradient descent phải phân kỳ khi
  $t > 2/L$; SGD với bước hằng phải dừng cách $\fstar$ một khoảng dương; ghi lịch
  sử dày hay thưa phải cho cùng số lần đánh giá hàm mục tiêu.
  \note{\textbf{Dùng khi bị hỏi làm sao biết cài đặt đúng.} Ý cần truyền đạt:
  bài toán có sẵn nghiệm đóng và biểu thức giải tích, nên nhóm em không phải tin
  vào việc đọc lại mã nguồn mà đối chiếu được với mốc tính ra bằng tay.

  Ba phép thử về phương pháp luận đáng nói riêng, vì chúng bắt thuật toán phải
  có hành vi \emph{xấu} đúng như lý thuyết dự đoán. Nếu GD không phân kỳ khi
  $t > 2/L$, hoặc SGD bước hằng lại hội tụ về đúng $\fstar$, thì có gì đó sai
  chứ không phải cài đặt tốt.

  Kể hai lỗi thật đã bắt được, vì cả hai đều để chương trình chạy trót lọt rồi
  trả về một con số trông hợp lý. Một, quy tắc nhận diện cột số có đơn vị biến
  \texttt{model\_36} thành số 36, làm mất cột định tính nhiều mức nhất. Hai,
  momentum của AGD tính theo $1/L$ nhưng ghép với bước do line search chọn, làm
  hàm mục tiêu bùng lên $\num{1.47e4}$. Không có mốc giải tích thì cả hai còn
  nguyên trong kết quả.}
\end{frame}
