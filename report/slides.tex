  \end{itemize}
  \note{\textbf{1:10--2:20.} Dữ liệu lấy từ Kaggle, một triệu bản ghi và 28 cột,
  không thiếu giá trị và không trùng lặp, nên phần chuẩn bị chỉ còn mã hóa với

  Sau mã hóa thì $d = 280$, trong đó 210 cột là tích từng cặp. Biến mục tiêu lấy

  \textbf{Nếu bị hỏi lấy mẫu có làm hỏng kết quả không:} không, với các kết luận

\begin{frame}{Hàm mục tiêu và ba con số quyết định độ khó}
  \[
    \min_{w \in \R^{d}} \;
    f(w) = \frac{1}{2n}\norm{Xw-y}_{2}^{2} + \frac{\lambda}{2}\norm{w}_{2}^{2},
    \quad
    L = \lambda_{\max}\Bigl(\tfrac{1}{n}X\trans X\Bigr) + \lambda,
    \;
    \mu = \lambda_{\min}\Bigl(\tfrac{1}{n}X\trans X\Bigr) + \lambda,
    \;
    \kappa = L/\mu
  \]
  \begin{columns}[T]
    \begin{column}{0.34\textwidth}
      \begin{tabular}{lr}
        \toprule
        $\lambda$  & \num{0.01} \\
        $L$        & \num{2.68309} \\
        $\mu$      & \num{0.01} \\
        $\kappa$   & \num{268.309} \\
        $\fstar$   & \num{0.0237430} \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.62\textwidth}
      Hessian $\tfrac{1}{n}X\trans X + \lambda I$ không chứa $w$, nên bài toán
      lồi mạnh và có nghiệm đóng $\wstar$. Nhờ vậy $\fstar$ tính được trước khi
      chạy thuật toán nào, và mọi biểu đồ vẽ sai số tuyệt đối $f(w_k) - \fstar$
      trên thang logarit.
      \medskip

      Ma trận Gram có hạng 273 trên 280 và trị riêng kế tiếp chỉ cỡ
      $2 \cdot 10^{-8}$, tức dữ liệu gần như không đóng góp gì vào $\mu$. Do đó
      $\mu = \lambda$ và $\kappa = L/\lambda$: độ khó của bài toán là thứ người
      làm chọn chứ không phải thứ dữ liệu áp đặt.
    \end{column}
  \end{columns}
  \note{\textbf{2:20--3:40.} Hàm mục tiêu gồm hai số hạng kéo về hai phía, một
  ép mô hình bám dữ liệu và một ép trọng số nhỏ lại, còn $\lambda$ định giá quy
  đổi giữa hai bên. Hai hằng số đứng trước thuần túy kỹ thuật: chia đôi để đạo
  hàm không kéo theo số 2, chia cho $n$ để $L$ không phụ thuộc cỡ mẫu.

  Ba con số bên trái quyết định toàn bộ phần sau. Cách hình dung mà nhóm em dùng
  suốt bài: hàm mục tiêu là một cái bát, $L$ là độ cong lớn nhất, $\mu$ là độ
  cong nhỏ nhất, $\kappa$ là độ méo. $\kappa = 1$ thì bát tròn xoay, thả bi phát
  xuống đáy luôn. $\kappa$ lớn thì bát bị bóp dẹt thành cái máng dài, hòn bi lăn
  ngang rất nhanh rồi bò dọc máng chậm hơn nhiều bậc, và toàn bộ khó khăn của
  các phương pháp bậc một nằm ở chỗ đó.

  Hai điểm bên phải đáng nhấn. Một, Hessian hằng nên có nghiệm đóng, và $\fstar$
  biết trước tới sai số máy. Đó là điều kiện làm nên toàn bộ phần đo đạc, vì sai
  số đo bằng thước tuyệt đối chứ không so với giá trị tốt nhất quan sát được.
  Hai, ma trận Gram suy biến nên $\mu$ trùng khít $\lambda$, tức chính hệ số
  hiệu chỉnh điều khiển số điều kiện. Phần năm khai thác đúng tính chất này.}
\end{frame}

\begin{frame}{Cách đo một lời giải tốt: hai thước chưa gặp nhau}
  \[
    r_i = \log \hat{p}_i - \log p_i,
    \qquad
    \mathrm{RMSE} = \sqrt{\frac{1}{m}\sum_{i=1}^{m} r_i^{2}},
    \qquad
    \delta = e^{\mathrm{RMSE}} - 1
  \]
  \begin{itemize}
    \item Phần dư nằm trên thang logarit nên RMSE đo độ lệch nhân tính, còn
          $\delta$ đọc nó theo tỉ lệ phần trăm của giá. Với RMSE \num{0.20601}
          thì $\delta = \num{0.2288}$, tức dự đoán vênh \num{22.9}\% và trên một
          chiếc máy giá trung vị \num{18555} thì vênh khoảng \num{4245} đơn vị
          tiền.
    \item Người chấm thuật toán đọc $f(w_k) - \fstar$, còn cửa hàng đọc sai lệch
          giá. Hai đại lượng ấy chưa gặp nhau ở chỗ nào, nên câu hỏi giải tới
          mức nào thì dừng vẫn còn treo.
  \end{itemize}
  \note{\textbf{3:40--4:30.} Lấy logarit biến mục tiêu thì đổi luôn cách đọc sai
  số, nên slide này chỉ làm một việc là bắc cầu giữa hai đơn vị.

  RMSE trên thang logarit không đọc thẳng ra tiền được. Công thức thứ ba quy nó
  về tỉ lệ: RMSE \num{0.20601} ứng với vênh \num{22.9}\% so với giá thật, và
  trên chiếc máy giá trung vị \num{18555} thì vênh khoảng \num{4245} đơn vị
  tiền. Con số \num{4245} này là mốc để so mọi con số khác trong bài, nên mong
  thầy cô nhớ giúp nhóm em.

  Phép quy đổi chỉ gần đúng, vì RMSE lấy trung bình của bình phương nên nặng hơn
  độ lệch trung bình. Nó đủ để so hai mô hình với nhau và bài này không cần hơn
  thế.

  Ý cuối slide dẫn sang phần hai. Mọi biểu đồ hội tụ đo bằng $f(w_k) - \fstar$,
  còn cửa hàng chỉ quan tâm sai số giá, mà hai thước đó chưa nối vào nhau. Phần
  hai nối chúng bằng một thí nghiệm riêng, và kết quả của nó đặt ngưỡng dừng cho
  toàn bộ phần còn lại.}
\end{frame}

\begin{frame}{Năm thuật toán đem so}
  \begin{center}
    \small
    \begin{tabular}{llll}
      \toprule
      Thuật toán & Thông tin dùng & Chi phí mỗi vòng & Tham số phải chọn \\
      \midrule
      Gradient descent & gradient tại $w_k$          & $\bigO(nd)$             & $t$, hoặc $\alpha$ và $\beta$ \\
      SGD              & gradient trên lô $B$ điểm   & $\bigO(Bd)$             & $B$, quy tắc chọn $\eta_k$ \\

      Newton           & toàn bộ Hessian             & $\bigO(nd^{2} + d^{3})$ & không có \\
      \bottomrule
    \end{tabular}
  \end{center}
  Hai cột cuối là hai thứ phân biệt năm thuật toán: mỗi phương pháp dùng bao
  nhiêu thông tin về $f$, và trả giá bao nhiêu cho mỗi vòng lặp. Với $d = 280$,
  chênh lệch chi phí đủ lớn để thứ hạng theo số vòng lặp và thứ hạng theo thời
  gian chạy không còn trùng nhau.
  \note{\textbf{4:30--5:00.} Bảng này chỉ để thầy cô đọc được phần ba mà chưa
  cần phần cài đặt chi tiết, nên nhóm em đọc theo cột chứ không theo dòng.

  Cột thứ hai xếp năm thuật toán theo lượng thông tin dùng, từ chỉ gradient tại

  ấy. Newton dùng nhiều nhất nên cần ít vòng lặp nhất, nhưng mỗi vòng đắt hơn
  một vòng gradient theo bậc $d^{2}$.

  Chính chỗ đó là lý do bài này phải vẽ hai biểu đồ cho mỗi so sánh, một theo
  vòng lặp và một theo giây. Nếu chi phí mỗi vòng như nhau thì hai biểu đồ chỉ
  là một hình vẽ trên hai thang, còn ở đây chúng cho hai thứ hạng khác nhau.

\end{frame}

\begin{frame}{RMSE ngừng cải thiện từ đâu}

      153 điểm ghi của sáu cấu hình hội tụ, tính trên một chiếc máy giá trung vị
      \num{18555}. Sai lệch giá giảm chậm hơn sai số tối ưu hóa: khớp hàm lũy
      thừa cho số mũ \num{0.68}, tức cắt sai số đi 10 lần chỉ giảm sai lệch giá
      khoảng 5 lần.
    \end{column}
  \end{columns}
  \note{\textbf{5:00--6:15.} Đây là hình trả lời câu hỏi treo ở slide trước, và
  cách đọc nó hơi ngược: trục hoành chạy từ sai số lớn bên trái sang sai số nhỏ
  bên phải, tức theo chiều các thuật toán tiến.

  Sáu đường đều phẳng lại và nằm chồng lên mức của nghiệm đóng, nhưng chúng
  phẳng ở những chỗ khác nhau. Ở cùng một mức sai số tối ưu hóa, các thuật toán
  không cho cùng một RMSE, vì $f(w_k) - \fstar$ chỉ là một số vô hướng còn hai
  điểm cùng giá trị hàm mục tiêu vẫn nằm ở hai chỗ khác nhau trong không gian
  tham số. Vì thế ngưỡng phải lấy theo trường hợp xấu nhất trong cả sáu đường,
  và đó chính là bảng bên phải.

  Bảng đọc theo dòng: dưới $10^{-4}$ thì sai lệch giá không quá \num{12.4} đơn
  vị tiền, dưới $10^{-6}$ thì không quá \num{0.61} đơn vị. Nhóm em khớp một hàm
  lũy thừa vào 85 điểm ghi trải từ $10^{-12}$ tới $10^{-1}$ và được số mũ
  \num{0.68}, nghĩa là mỗi chữ số chính xác thêm về phía tối ưu hóa mua được
  ngày càng ít về phía cửa hàng.

  \textbf{Nếu bị hỏi vì sao số mũ không phải \num{0.5} hay 1:} vì $f$ bậc hai
  nên sai số hàm tỉ lệ với bình phương khoảng cách tới $\wstar$, và hai bức

\section{Ba cơ chế giải thích chênh lệch}

\begin{frame}{Cơ chế thứ nhất: độ dài bước so với ngưỡng $2/L$}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}

    \end{column}
    \begin{column}{0.46\textwidth}

        \midrule

        \bottomrule
      \end{tabular}
      \medskip

      Sai số theo hướng riêng ứng với $\lambda_i$ nhân với
      $\abs{1 - t\lambda_i}$ mỗi vòng, nên hệ số của hướng ứng với
      $\lambda_{\max} = L$ vượt 1 ngay khi $t$ vượt $2/L = \num{0.745}$, còn
      đúng tại ngưỡng thì nó bằng 1 và hướng đó đứng yên.
    \end{column}
  \end{columns}
  \note{\textbf{12:15--13:20.} Cơ chế thứ nhất trả lời câu vì sao ba phương pháp
  bậc một bám nhau ở đầu và vì sao một cấu hình về đích sớm lại không đi xa
  được.

  Mời thầy cô nhìn đường đi lên ở phía trên biểu đồ. Đó là $t = 2.1/L$, nhóm em
  cố ý cho chạy vượt ngưỡng, và nó đẩy hàm mục tiêu lên $5{,}4 \cdot 10^{11}$.
  Ngay cạnh đó, $t = 1.9/L$ hội tụ sau 20 vòng lặp. Hai cấu hình chênh nhau 10\%
  quanh $2/L$ mà kết cục lật hẳn, nên đây là ranh giới thật chứ không phải cận
  an toàn nới rộng.

  Cơ chế nằm ở từng hướng riêng, viết ở cuối cột phải. Sau mỗi vòng, sai số theo
  hướng ứng với trị riêng $\lambda_i$ nhân với $\abs{1 - t\lambda_i}$. Hướng dốc
  nhất có $\lambda_i = L$, nên $t$ vượt $2/L$ là hệ số vượt 1 và thành phần đó
  nhân lên mỗi vòng thay vì tắt đi. Đúng tại ngưỡng, hệ số bằng 1, tức nó đứng

  chạm vào nó.

\begin{frame}{Chi phí của backtracking}
  \begin{center}
    \begin{tabular}{lrr}
      \toprule
      $\alpha = 0.5$ & Đánh giá $f$ mỗi vòng & Riêng phần line search \\
      \midrule
      $\beta = 0.5$ & \num{2.40} & \num{1.40} \\
      $\beta = 0.8$ & \num{3.45} & \num{2.45} \\
      $\beta = 0.9$ & \num{5.1}  & \num{4.1}  \\
      \bottomrule
    \end{tabular}
  \end{center}
  Trong chín cấu hình đã quét, số lần đánh giá hàm mỗi vòng lặp chỉ phụ thuộc
  $\beta$ và gần như không đổi khi $\alpha$ chạy từ \num{0.1} tới \num{0.5}. Đúng
  theo vai trò hai tham số trong điều kiện Armijo: $\beta$ quyết định mỗi lần
  thử thu nhỏ bước bao nhiêu nên nó định số lần thử, còn $\alpha$ chỉ dịch ngưỡng
  chấp nhận. Một lần đánh giá mỗi vòng là chi phí bắt buộc, nên cột phải mới là
  phần line search thật sự sinh ra.
  \note{\textbf{Dùng khi bị hỏi backtracking đắt bao nhiêu.} Bước cố định đòi
  biết $L$ trước. Không tính được $L$ thì thử: bắt đầu bằng một bước dài, thấy
  chưa giảm đủ thì nhân với $\beta < 1$, lặp tới khi điều kiện Armijo chấp nhận.
  Điều kiện ấy bắt bước hiện tại phải giảm được ít nhất $\alpha$ phần của mức
  giảm mà xấp xỉ tuyến tính hứa hẹn, và đòi hỏi đó cần thiết vì một bước rất nhỏ
  luôn làm $f$ giảm chút ít.

  Cặp $\beta$ này là ví dụ sạch nhất trong cả bài cho câu hỏi vì sao phải vẽ hai
  biểu đồ. Các cấu hình gần trùng nhau trên trục vòng lặp nhưng tách hẳn trên
  trục thời gian, vì chênh lệch nằm ở số lần đánh giá hàm bên trong mỗi vòng, mà
  trục vòng lặp không nhìn thấy chi phí đó.

  Chốt lại, backtracking chậm hơn bước cố định tốt nhất khoảng 21\%, và khoản
  chênh ấy mua lấy chỗ không phải biết $L$. Bài này $L$ tính dễ nên bước cố định
  hợp lý hơn, nhưng nhận xét ấy đảo chiều ngay khi $d$ lớn tới mức không tính
  nổi trị riêng lớn nhất của ma trận Gram.}
\end{frame}

\begin{frame}{Momentum phải khớp với độ dài bước thực tế}

        \item Khởi động lại đưa tổng thời gian từ \SI{6.67}{\second} còn

  ngoại suy trước rồi mới lấy gradient ở đó.

  \textbf{Nên kể chỗ này:} lần cài đặt đầu, nhóm em để line search chọn $t$
  nhưng vẫn tính momentum theo $1/L$, và hàm mục tiêu bùng lên
  $1{,}47 \cdot 10^{4}$. Chương trình chạy trót lọt, không báo lỗi, chỉ trả về
  một con số trông hợp lý, nên chỉ có đối chiếu với mốc giải tích mới bắt được
  loại lỗi này. Nhóm em giữ một phép kiểm trong bộ kiểm thử để nó không quay

  Momentum tăng dần gây dao động ở phần đuôi, nhìn thấy như gợn sóng trên biểu
  đồ, và điều kiện gradient nhân bước vừa đi lớn hơn không cắt đúng các dao động

\begin{frame}{SGD: kích thước lô khi mỗi lô có bước riêng}
  \begin{center}\resultgraphic[0.58]{sgd_batch_epoch}\end{center}
  Khi mỗi kích thước lô nhận $\eta_0 = 1/L_B$ riêng, năm cấu hình cho sai số
  cuối gần như nhau, từ $8{,}2 \cdot 10^{-4}$ tới $1{,}2 \cdot 10^{-3}$. Chênh
  lệch dồn hết sang thời gian chạy: $B = 1$ mất \SI{34.5}{\second} cho 30 lượt
  duyệt còn $B = 256$ chỉ mất \SI{1.86}{\second}, tức nhanh hơn 18 lần với cùng
  số lần chạm vào dữ liệu.
  \note{\textbf{Dùng khi bị hỏi vì sao chọn $B = 64$ hoặc $B = 256$.} Điểm cần
  làm rõ trước: biểu đồ này chạy với mỗi lô một bước riêng, khác hẳn thí nghiệm
  đối chứng dùng chung một bước ở phần chính. Khi bước đã chỉnh theo lô thì kích
  thước lô không còn quyết định độ chính xác, chỉ còn quyết định thời gian.

  Nói rõ chênh lệch 18 lần đến từ đâu, vì đây là chỗ dễ hiểu nhầm: lô nhỏ trả
  giá không phải vì tính nhiều hơn mà vì nó chia cùng khối lượng tính toán thành
  nhiều lần gọi hơn, mà chi phí cố định của mỗi lần gọi không giảm theo kích
  thước lô. Với $B = 1$ thì một lượt duyệt là \num{160000} lần cập nhật.

  Nếu bị hỏi tiếp con số 18 lần có phổ quát không thì trả lời là không, và ghi
  rõ đây là dự đoán chứ nhóm em chưa đo: cài bằng ngôn ngữ biên dịch, hoặc $d$
  lớn hơn nhiều để phần số học lấn át, thì tỉ lệ sẽ co lại.}
\end{frame}

\begin{frame}{Newton và L-BFGS: các biến thể}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \resultgraphic[0.48]{lbfgs_time}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item Ba biến thể Newton: phân rã lại mỗi vòng \SI{0.136}{\second}, dùng
              lại phân rã cũ \SI{0.133}{\second}, bản damped
              \SI{0.168}{\second}. Cả ba chỉ chạy một vòng nên khác biệt giữa
              chúng là sản phẩm của dạng bậc hai.
        \item L-BFGS với bốn kích thước bộ nhớ đều đạt $10^{-6}$ sau đúng 8 vòng
              lặp, tổng thời gian từ \SI{1.38}{\second} với $m = 10$ tới
              \SI{1.97}{\second} với $m = 3$.
        \item Bộ nhớ lớn hơn không đem lại lợi ích nào vì bài toán là hàm bậc
              hai và tám cặp hiệu đã đủ dựng lại phần phổ mà thuật toán cần, còn
              chênh lệch thời gian nằm trong khoảng dao động giữa các lần chạy.
      \end{itemize}
    \end{column}
  \end{columns}
  \note{\textbf{Dùng khi bị hỏi chi tiết về nhóm bậc hai.} Ba biến thể Newton
  trùng nhau trên trục vòng lặp và gần trùng trên trục thời gian, và cần giải
  thích đúng lý do: backtracking chấp nhận ngay $t = 1$ nên chỉ tốn thêm một lần
  đánh giá hàm, còn phân rã lại hay dùng lại thì vô nghĩa khi chỉ có một vòng
  lặp. Hai cặp này chỉ tách nhau trên bài toán phi tuyến.

  Với L-BFGS, chọn $m$ không quan trọng ở bài này. Cả bốn cấu hình đạt $10^{-6}$
  sau đúng 8 vòng lặp, vì trong 8 vòng đó số cặp tích lũy được còn ít hơn cả
  $m = 10$, và vì hàm bậc hai không đòi nhiều cặp hơn thế.

  \textbf{Nếu bị hỏi vì sao dùng Cholesky chứ không phải nghịch đảo tường minh:}
  nghịch đảo kém ổn định về số học và tốn nhiều phép tính hơn phân rã rồi giải
  thế.}
\end{frame}

\begin{frame}{Quy đổi hàm mục tiêu của thư viện}
  \begin{equation*}
    f(w) = \frac{1}{2n}\norm{Xw-y}_{2}^{2} + \frac{\lambda}{2}\norm{w}_{2}^{2}
  \end{equation*}
  \begin{itemize}
    \item \texttt{Ridge} cực tiểu hóa
          $\norm{Xw-y}_{2}^{2} + \alpha\norm{w}_{2}^{2}$, tức thiếu cả
          $\tfrac{1}{2n}$ lẫn $\tfrac{\lambda}{2}$. Nhân hàm mục tiêu với $2n$
          không đổi nghiệm, nên $\alpha = \lambda n = \num{1600}$.
    \item \texttt{SGDRegressor} với \texttt{penalty="l2"} cực tiểu hóa đúng dạng
          trên, nên $\alpha = \lambda$.
    \item \texttt{LinearRegression} không có số hạng phạt nên nó ứng với
          $\lambda = 0$ và giải một bài toán khác, đưa vào chỉ để có mốc tham
          chiếu.
  \end{itemize}
  Nhóm kiểm phép quy đổi bằng số: nghiệm mà \texttt{Ridge} với solver
  \texttt{cholesky} trả về cách nghiệm đóng $1{,}5 \cdot 10^{-15}$, sai số tương
  đối $1{,}9 \cdot 10^{-15}$, và chênh lệch hàm mục tiêu bằng đúng 0 trong số
  học dấu phẩy động.
  \note{\textbf{Dùng khi bị hỏi so sánh với thư viện có công bằng không.} Trả
  lời bằng con số: $\alpha = \lambda n = \num{1600}$ cho \texttt{Ridge}, còn
  \texttt{SGDRegressor} cực tiểu hóa đúng dạng của nhóm em nên
  $\alpha = \lambda$.

  Nhấn vào cách kiểm chứng chứ không chỉ vào công thức: nhóm em viết phép kiểm
  này thành một khẳng định trong bộ kiểm thử thay vì một dòng in ra để đọc bằng
  mắt, vì mọi con số ở bảng so với thư viện sẽ vô nghĩa nếu nó thất bại.}
\end{frame}

\begin{frame}{Kiểm thử tính đúng đắn của cài đặt}
  60 phép thử tự động chạy trước khi khảo sát tham số. Bốn phép đối chiếu thẳng
  với mốc giải tích:
  \begin{itemize}
    \item gradient giải tích so với sai phân trung tâm tại
          $\eps = 10^{-6}$, sai số tương đối dưới $10^{-6}$;
    \item Hessian so với sai phân của gradient, cùng ngưỡng;
    \item nghiệm đóng là điểm dừng, $\norm{\nabla f(\wstar)} < 10^{-12}$;
    \item mười cấu hình tất định cùng hội tụ về một nghiệm trên bài toán tổng
          hợp $n = 100$, $d = 5$.
  \end{itemize}
  Bốn phép thử còn lại kiểm chính cách đo: gradient descent phải phân kỳ khi
  $t > 2/L$, SGD bước hằng phải dừng cách $\fstar$ một khoảng dương, momentum
  phải tính theo độ dài bước thực tế, và ghi lịch sử dày hay thưa phải cho cùng
  số lần đánh giá hàm.
  \note{\textbf{Dùng khi bị hỏi làm sao biết cài đặt đúng.} Ý cần truyền đạt:
  bài toán có sẵn nghiệm đóng và có sẵn biểu thức giải tích, nên nhóm em không
  phải tin vào việc đọc lại mã nguồn mà đối chiếu được với mốc tính ra bằng tay.

  Bốn phép thử về phương pháp luận đáng nói riêng, vì chúng bắt thuật toán phải
  có hành vi \emph{xấu} đúng như lý thuyết dự đoán. Nếu gradient descent không
  phân kỳ khi $t > 2/L$, hoặc SGD bước hằng lại hội tụ về đúng $\fstar$, thì có
  gì đó sai chứ không phải cài đặt tốt.

  Kể ba lỗi thật đã bắt được, vì cả ba đều để chương trình chạy trót lọt rồi trả
  về một con số trông hợp lý. Một, quy tắc nhận diện cột số có đơn vị biến cột
  \texttt{model} thành số và làm mất biến định tính nhiều mức nhất. Hai, momentum
  tính theo $1/L$ nhưng ghép với bước do line search chọn, làm hàm mục tiêu bùng
  lên $1{,}47 \cdot 10^{4}$. Ba, thiếu tiêu chí dừng theo đình trệ, khiến một
  lần chạy tiêu 98,7\% thời gian vào phần đã hết cải thiện.}
\end{frame}
