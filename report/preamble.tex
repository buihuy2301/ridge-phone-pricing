% ---------------------------------------------------------------------------
% Figure inclusion that tolerates figures which have not been generated yet.
% #1 = file stem in results/figures, #2 = caption, #3 = label
% ---------------------------------------------------------------------------
% The figure stems contain underscores, so the name has to be detokenized
% before it is typeset.
\newcommand{\missingfigbox}[1]{%
  \fbox{\parbox[c][3.2cm][c]{0.86\linewidth}{\centering
    \texttt{\detokenize{#1}.pdf}\par\vspace{4pt}
    \small chưa được sinh ra từ notebook\par}}%
}

% The path is spelled out rather than left to \graphicspath, which lists the
% slide directory first and would otherwise hand the report the wide, reduced
% copies meant for Beamer.
% Placement is [tp] rather than [htbp] on purpose. With 'h' a figure declared
% between two paragraphs is typeset right there and cuts the argument in half,
% and with 'b' it lands under a paragraph that continues on the next page, which
% reads the same way. Top of page or a float page keeps the prose continuous.
\newcommand{\resultfig}[3]{%
  \begin{figure}[tp]
    \centering
    \IfFileExists{../results/figures/#1.pdf}
      {\includegraphics[width=0.8\linewidth]{../results/figures/#1.pdf}}
      {\missingfigbox{#1}}
    \caption{#2}
    \label{#3}
  \end{figure}%
}

% Figures are 6 by 4 inches, so a full-width copy is too tall for a slide that
% also carries text. The height is capped as a fraction of the text height and
% the aspect ratio is preserved; crowded frames pass a smaller fraction.
\newcommand{\resultgraphic}[2][0.72]{%
  \IfFileExists{../results/figures/slides/#2.pdf}
    {\includegraphics[width=\linewidth,height=#1\textheight,keepaspectratio]{../results/figures/slides/#2.pdf}}
    {\IfFileExists{../results/figures/#2.pdf}
      {\includegraphics[width=\linewidth,height=#1\textheight,keepaspectratio]{../results/figures/#2.pdf}}
      {\missingfigbox{#2}}}%
}
